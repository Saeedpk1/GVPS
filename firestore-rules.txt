rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function hasSettings() { return exists(/databases/$(database)/documents/settings/main); }
    function isOwner() { return signedIn() && hasSettings() && get(/databases/$(database)/documents/settings/main).data.ownerUid == request.auth.uid; }
    function staffDoc() { return get(/databases/$(database)/documents/staff/$(request.auth.uid)).data; }
    function isTeacher() { return signedIn() && exists(/databases/$(database)/documents/staff/$(request.auth.uid)) && staffDoc().active == true; }
    function myClass(c) { return isTeacher() && c in staffDoc().classes; }

    // School settings. The first person to sign in becomes the owner.
    match /settings/main {
      allow read: if signedIn();
      allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid;
      allow update: if isOwner() || (signedIn() && !('ownerUid' in resource.data) && request.resource.data.ownerUid == request.auth.uid);
    }
    // Teachers: only the owner adds or changes them. A teacher can read their own record.
    match /staff/{uid} {
      allow read: if isOwner() || (signedIn() && request.auth.uid == uid);
      allow write: if isOwner();
    }
    // Students: teachers can add and edit students of their own classes only.
    match /students/{id} {
      allow read: if isOwner() || isTeacher();
      allow create: if isOwner() || myClass(request.resource.data.cls);
      allow update: if isOwner() || (myClass(resource.data.cls) && myClass(request.resource.data.cls));
      allow delete: if isOwner();
    }
    // Fee receipts: teachers can add receipts in their own name. Only the owner can cancel or mark cash received.
    match /payments/{id} {
      allow read: if isOwner() || isTeacher();
      allow create: if isOwner() || (isTeacher() && request.resource.data.byUid == request.auth.uid);
      allow update, delete: if isOwner();
    }
    // Attendance: teachers can mark their own classes only.
    match /attendance/{id} {
      allow read: if isOwner() || isTeacher();
      allow write: if isOwner() || myClass(request.resource.data.cls);
    }
    // Extra charges: owner only.
    match /charges/{id} {
      allow read: if isOwner() || isTeacher();
      allow write: if isOwner();
    }
  }
}
